cmake_minimum_required(VERSION 3.22)
project(comicarchive C)

if(NOT EXISTS "${ARCHIVE_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set ARCHIVE_SOURCE to the prepared libarchive 3.8.11 source directory")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(POSIX_REGEX_LIB NONE CACHE STRING "" FORCE)
foreach(feature MBEDTLS NETTLE OPENSSL MD LIBB2 LZ4 LZO LZMA ZSTD BZip2 LIBXML2
        BSDXML EXPAT WIN32_XMLLITE PCREPOSIX PCRE2POSIX LIBGCC CNG TAR CPIO CAT UNZIP
        XATTR ACL ICONV TEST COVERAGE INSTALL)
    set(ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(ENABLE_ZLIB ON CACHE BOOL "" FORCE)
add_subdirectory("${ARCHIVE_SOURCE}" libarchive EXCLUDE_FROM_ALL)

add_library(comicarchive SHARED rar_jni.c)
target_compile_definitions(comicarchive PRIVATE LIBARCHIVE_STATIC _FILE_OFFSET_BITS=64)
target_include_directories(comicarchive PRIVATE "${ARCHIVE_SOURCE}/libarchive")
if(JNI_HEADER_ROOT)
    target_include_directories(comicarchive PRIVATE "${JNI_HEADER_ROOT}")
    if(WIN32)
        target_include_directories(comicarchive PRIVATE "${JNI_HEADER_ROOT}/win32")
    else()
        target_include_directories(comicarchive PRIVATE "${JNI_HEADER_ROOT}/linux")
    endif()
endif()
target_link_libraries(comicarchive PRIVATE archive_static)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(archive_static PRIVATE -ffunction-sections -fdata-sections)
    target_compile_options(comicarchive PRIVATE -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
    target_link_options(comicarchive PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
endif()
if(ANDROID)
    target_link_options(comicarchive PRIVATE -Wl,-z,max-page-size=16384)
elseif(MINGW)
    target_link_options(comicarchive PRIVATE -static-libgcc)
    set_target_properties(comicarchive PROPERTIES PREFIX "")
endif()
